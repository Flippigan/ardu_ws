```cpp
// Instance ID for multi-vehicle DDS support
#ifndef AP_DDS_INSTANCE_ID
#define AP_DDS_INSTANCE_ID 2  // Instance 2 (3rd out of 5 UAVs)
#endif

// Helper macros for string concatenation
#define AP_DDS_STR_HELPER(x) #x
#define AP_DDS_STR(x) AP_DDS_STR_HELPER(x)
#define AP_DDS_INSTANCE_STR AP_DDS_STR(AP_DDS_INSTANCE_ID)

// Topic name generation macros
#define AP_DDS_TOPIC(name) "rt/ap_" AP_DDS_INSTANCE_STR "/" name
#define AP_DDS_SERVICE(name) "rs/ap_" AP_DDS_INSTANCE_STR "/" name
#define AP_DDS_REQUEST(name) "rq/ap_" AP_DDS_INSTANCE_STR "/" name
#define AP_DDS_REPLY(name) "rr/ap_" AP_DDS_INSTANCE_STR "/" name
```
